\begin{frame}
\frametitle{Module A: Bird Nest Detection --- Data \& Model}
\vspace{-5pt}
\begin{columns}[T]
\column{0.48\linewidth}
\textbf{Dataset \& Augmentation}
\begin{itemize}
  \item 54 COCO-format images (43 train, 11 val)
  \item Bounding boxes + instance segmentation polygons
  \item Deterministic augmentation pipeline:
    \begin{itemize}
      \item Custom Random IoU Crop (scale 0.3--1.0)
      \item Photometric jitter (brightness, contrast)
      \item ResizeMax + PadSquare ($512\times512$)
    \end{itemize}
  \item \textit{Why deterministic? Generative augmentation introduced physically implausible artifacts.}
\end{itemize}

\column{0.48\linewidth}
\textbf{Model \& Training Config}
\begin{itemize}
  \item Model: Mask R-CNN (ResNet50 + FPN v2)
  \item Input: $512\times512$, batch size 4, 40 epochs
  \item Optimizer: AdamW (lr=$5\times10^{-4}$) + OneCycleLR
  \item Mixed precision (AMP) enabled
  \item $\sim$27\,s per epoch; GPU utilization 70--85\%
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{Module C: Infrared High-Temperature Detection}
\vspace{-5pt}
\begin{columns}[T]
\column{0.48\linewidth}
\textbf{Approach: OCR + Pixel-Mapping Hybrid}
\begin{itemize}
  \item No dedicated DL model training required
  \item EasyOCR parses high/low temperature scale from legend
  \item Pixel intensity $\to$ temperature mapping with heuristic OCR correction
  \item User-defined alarm temperature threshold (default 50$^\circ$C)
  \item Outputs: hotspot boxes, per-hotspot temp., global max temp., annotated visualization
\end{itemize}

\column{0.48\linewidth}
\textbf{Suppression Techniques}
\begin{itemize}
  \item \textbf{Border watermark mask}: detects and masks bright white logo/watermark near image border
  \item \textbf{OCR text-region mask}: excludes legend and scale digits to reduce false positives
  \item Filters regions above alarm threshold only
\end{itemize}
\end{columns}
\end{frame}

\vspace{-5pt}
\begin{columns}[T]

\textbf{Flask REST API Features}
\begin{itemize}
  \item Batch image upload, on-demand inference
  \item Model selection: Bird Nest / Oil Leak / Infrared
  \item JSON result (boxes, confidence/temperature, class) + visualization
  \item Single \& ZIP download for offline archiving
  \item Statistical dashboards for inference history
  \item Large-image handling: resize / tiling to avoid OOM
  \item Gunicorn multi-worker for production concurrency
\end{itemize}

\column{0.48\linewidth}
\textbf{Architecture \& SaaS Role}
\begin{itemize}
  \item All three modules share one upload/inference/download API
  \item Configurable model registry: weight path managed per model type
  \item ``Model-as-a-Service'' foundation for multi-tenant RBAC
  \item Fallback to local checkpoint if registry entry missing
  \item Structured exception handling at API layer
\end{itemize}

\begin{frame}
\frametitle{Problems \& Challenges}
\vspace{-5pt}
\begin{block}{1. Generalization}
\begin{itemize}
  \item Bird nest AP$_{50}$=1.0 on same-distribution validation --- may drop in real conditions (haze, backlight, new tower types, seasonal variation)
  \item Cross-scene data, negative and hard-sample design needed as platform-level requirements
\end{itemize}
\end{block}
\begin{block}{2. Technical Choices}
\begin{itemize}
  \item \textbf{SaaS backend}: feature-complete prototype (RBAC, task queue) vs.\ production-grade auth (OAuth2, audit logs) --- current preference: demonstrable multi-model prototype first
  \item \textbf{Oil leak}: YOLOv8 box-level sufficient for location; pixel-level segmentation (U-Net) needed only if area statistics required
  \item \textbf{Infrared}: standalone module sufficient now; RGB+heatmap fusion possible for multi-modal analysis
\end{itemize}
\end{block}
\begin{block}{3. Data \& Deployment Constraints}
\begin{itemize}
  \item Dataset size modest (54 bird nest; $\sim$333 merged oil leak); annotation quality critical
  \item Flask multi-process handles moderate load; queue+worker pool needed for high concurrency
\end{itemize}
\end{block}
\end{frame}
