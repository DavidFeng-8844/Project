\frametitle{1. Background \& Motivation}
\begin{block}{The Industry Gap}
While deep learning algorithms for power grid inspection achieve high theoretical accuracy, transitioning them from \textbf{isolated Python scripts} to \textbf{multi-tenant industrial deployments} remains a critical engineering bottleneck.
\end{block}
\vspace{0.3cm}
\begin{columns}[T]
\column{0.48\linewidth}
\textbf{Challenges Overcome:}
\begin{itemize}
    \item \textbf{Data Constraints:} Optimizing models on extremely small, highly biased datasets.
    \item \textbf{Hardware Constraints:} Serving multiple heavy DL models concurrently under strict edge VRAM limits (12GB).
\end{itemize}
\column{0.48\linewidth}
\textbf{Project Deliverables:}
\begin{itemize}
    \item 3 High-accuracy detection modules (RGB \& Infrared).
    \item A fully decoupled, multi-tenant Flask SaaS platform.
    \item Zero-latency asynchronous task queue architecture.
\end{itemize}
\end{columns}
\end{frame}

\frametitle{7. SaaS Innovation 1: Async Event-Driven Dispatch}
\begin{columns}[T]
\column{0.45\linewidth}
\textbf{Synchronous vs. Asynchronous}
\begin{itemize}
    \item \textbf{Sync:} Blocks the thread. 20 concurrent users = cascading 504 Gateway Timeouts.
    \item \textbf{Async:} \texttt{ThreadPoolExecutor} instantly assigns tasks to memory queue.
\end{itemize}
\vspace{0.3cm}
\textbf{Evaluation Results}
\begin{itemize}
    \item Client receives HTTP 202 in \textbf{$<$68ms}.
    \item Hardware achieves sustained throughput of \textbf{124 images/min}.
\end{itemize}
\column{0.5\linewidth}
\centering
\includegraphics[width=0.9\textwidth]{saas/figures/plot_async_vs_sync.png}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{8. SaaS Innovation 2: Lazy-Load Model Registry}
\begin{columns}[T]
\column{0.45\linewidth}
\textbf{Target Environment Limits}
\begin{itemize}
    \item Edge Node VRAM ceiling: 12GB.
    \item Mask R-CNN (5.5GB) + YOLOv8s (1.5GB) loaded globally causes startup Out-of-Memory (OOM).
\end{itemize}
\vspace{0.3cm}
\textbf{Registry Pattern}
\begin{itemize}
    \item Weights are \textit{only} loaded upon absolute first invocation.
    \item Memory flatlines securely at \textbf{7.4GB}.
    \item Zero memory leakage observed over extended stress testing.
\end{itemize}
\column{0.5\linewidth}
\centering
\includegraphics[width=0.9\textwidth]{saas/figures/plot_memory_stability.png}
\end{columns}

\frametitle{9. System Security \& Automated Validation}
\begin{columns}[T]
\column{0.45\linewidth}
\textbf{Zero-Trust Tenant Isolation}
\begin{itemize}
    \item Eschewed implicit ORM hooks.
    \item Explicit predicate injection at routing layer: \texttt{.filter\_by(tenant\_id=jwt)}
\end{itemize}
\vspace{0.3cm}
\textbf{Empirical Attack Matrix}
\begin{itemize}
    \item 150 simulated cross-tenant assaults (forged IDs, aggregate enumeration).
    \item Result: \textbf{0\% Data Leakage}. Responses mathematically forced to 404 or \texttt{[]}.
\end{itemize}
