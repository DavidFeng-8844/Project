% Slide 3: Methodology (Data)
\begin{frame}{Methodology (1): Robust Data Pipeline}
    \framesubtitle{The ``Generative Trap'' \& The Deterministic Solution}
    
    \begin{alertblock}{Issue}
        Generative models introduced physically implausible artifacts (e.g., rust on non-metallic parts).
    \end{alertblock}

    \begin{block}{Solution: Deterministic Augmentation}
        Implemented using \texttt{torchvision.transforms.v2}:
        \begin{itemize}
            \item \textbf{Custom Random IoU Crop:} Forces model to learn from partial object views (Scales: 0.3--1.0).
            \item \textbf{Photometric Distortions:} Simulates field conditions (Brightness $\pm 12.5\%$, Contrast $\pm 50\%$).
            \item \textbf{Geometric Regularization:} Consistent aspect ratios via \texttt{ResizeMax} and \texttt{PadSquare}.
        \end{itemize}
    \end{block}
\end{frame}

% Slide 9: Discussion
\begin{frame}{Discussion: Key Engineering Decisions}
    \begin{columns}
        \begin{column}{0.5\linewidth}
            \begin{block}{Mask R-CNN vs. YOLO}
                \begin{itemize}
                    \item \textbf{Why Mask R-CNN?} Instance Segmentation is critical for damage assessment (pixel-level).
                    \item \textbf{Trade-off:} 100-200ms latency is acceptable for post-flight analysis.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\linewidth}
            \begin{block}{Generative vs. Deterministic}
                \begin{itemize}
                    \item \textbf{Failure Mode:} Stable Diffusion generated ``noise patterns'' instead of semantic nest features (Feature Leakage).
                    \item \textbf{Pivot:} Deterministic augmentation ensures physical fidelity and traceability.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
